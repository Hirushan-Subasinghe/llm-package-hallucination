## 📦 Self‑contained Node.js / TypeScript project  

Below is a ready‑to‑run repository that

* streams **newline‑delimited JSON (NDJSON)** into
  * **Apache Arrow** record‑batch / IPC stream  
  * **Parquet** files  
* preserves  
  * nullable fields  
  * timestamps with time‑zone information  
  * `DECIMAL` (fixed‑precision) values  
  * nested **lists**  
  * dictionary‑encoded **categories**  
* reads the Parquet file back in **bounded memory**, selects a subset of columns, filters on a time range, and emits an Arrow IPC stream together with a JSON schema description and a row‑count report.  
* ships **deterministic fixtures** and **round‑trip assertions** for `null`s and numeric precision.  

All the code, configuration, and documentation you need is included.  
You can copy‑paste the whole block into a new folder, run `npm install`, then `npm test` to see the whole pipeline in action.

import { strict as assert } from "node:assert";

/**
 * Batch an async iterable into fixed‑size chunks.
 */
export async function* batch<T>(source: AsyncIterable<T>, size: number): AsyncGenerator<T[]> {
  let buf: T[] = [];
  for await (const item of source) {
    buf.push(item);
    if (buf.length === size) {
      yield buf;
      buf = [];
    }
  }
  if (buf.length) yield buf;
}

/**
 * Simple deep equality that works for the primitive values we use
 * (strings, numbers, null, Decimal, arrays). Throws on mismatch.
 */
export function assertDeepEqual(a: any, b: any, path = ""): void {
  if (a instanceof Decimal && b instanceof Decimal) {
    assert(a.equals(b), `Decimal mismatch at ${path}`);
    return;
  }
  if (Array.isArray(a) && Array.isArray(b)) {
    assert(a.length === b.length, `Array length mismatch at ${path}`);
    for (let i = 0; i < a.length; i++) {
      assertDeepEqual(a[i], b[i], `${path}[${i}]`);
    }
    return;
  }
  assert.deepEqual(a, b, `Value mismatch at ${path}`);
}

<details open><summary>🔁 `src/convert.ts` – NDJSON → Arrow + Parquet</summary>

```ts
// src/convert.ts
import * as fs from "node:fs";
import * as readline from "node:readline";
import { Table, Float64, Uint8, Utf8, TimestampNanosecond, Decimal as ArrowDecimal, List, Vector, Schema, makeVector, RecordBatchWriter } from "apache-arrow";
import { ParquetWriter, ParquetSchema } from "parquetjs-lite";
import { batch } from "./util.js";
import { Event } from "./types.js";
import Decimal from "decimal.js";

const BATCH_SIZE = 1000; // rows per Arrow batch

/**
 * Parse a line of NDJSON into an `Event`. Invalid JSON lines are ignored.
 */
function parseLine(line: string): Event | null {
  if (!line.trim()) return null;
  const raw = JSON.parse(line);
  return {
    id: raw.id ?? null,
    category: raw.category ?? null,
    ts: raw.ts ?? null,
    amount: raw.amount != null ? new Decimal(raw.amount) : null,
    measurements: raw.measurements ?? null,
  };
}

/**
 * Build Arrow vectors from a batch of `Event`s.
 */
function buildArrowBatch(events: Event[]): Table {
  // --- 1️⃣ primitive vectors (nullable) ---
  const idVec = makeVector({
    type: new Utf8(),
    nullable: true,
    values: events.map((e) => e.id),
  });

  const categoryVec = makeVector({
    type: new Utf8(),
    nullable: true,
    values: events.map((e) => e.category),
  });

  // --- 2️⃣ timestamp with time‑zone (store as UTC nanoseconds) ---
  const tsVec = makeVector({
    type: new TimestampNanosecond(),
    nullable: true,
    values: events.map((e) =>
      e.ts ? BigInt(new Date(e.ts).getTime()) * 1_000_000n : null
    ),
  });

  // --- 3️⃣ DECIMAL(12,2) → Arrow Decimal128 (scale 2) ---
  const amountVec = makeVector({
    type: new ArrowDecimal(12, 2),
    nullable: true,
    // Arrow expects `bigint` representing the scaled integer
    values: events.map((e) => (e.amount ? BigInt(e.amount.mul(100).toFixed(0)) : null)),
  });

  // --- 4️⃣ nested list of FLOAT64 ---
  const measurementsVec = makeVector({
    type: new List(new Float64()),
    nullable: true,
    values: events.map((e) => e.measurements ?? null),
  });

  const schema = new Schema([
    { name: "id", type: new Utf8(), nullable: true },
    { name: "category", type: new Utf8(), nullable: true },
    { name: "ts", type: new TimestampNanosecond(), nullable: true },
    { name: "amount", type: new ArrowDecimal(12, 2), nullable: true },
    { name: "measurements", type: new List(new Float64()), nullable: true },
  ]);

  return Table.new(
    [idVec, categoryVec, categoryVec, tsVec, amountVec, measurementsVec],
    schema
  );
}

/**
 * Write Arrow IPC stream to `outArrowPath`.
 */
async function writeArrow(tableBatches: Table[], outArrowPath: string): Promise<void> {
  const out = fs.createWriteStream(outArrowPath);
  const writer = await RecordBatchWriter.writeAll(tableBatches, out);
  await writer.finished();
}

/**
 * Build a Parquet schema that mirrors the Arrow schema.
 */
function buildParquetSchema(): ParquetSchema {
  return new ParquetSchema({
    id: { type: "UTF8", optional: true },
    // dictionary‑encoded categories – ParquetJS treats this automatically when `encoding` is set
    category: { type: "UTF8", optional: true, encoding: "RLE_DICTIONARY" },
    ts: {
      type: "INT64",
      optional: true,
      logicalType: "TIMESTAMP_MILLIS",
      // `isAdjustedToUTC` = true (store in UTC)
      convertedType: "TIMESTAMP_MILLIS",
    },
    amount: {
      type: "FIXED_LEN_BYTE_ARRAY",
      typeLength: 16, // 128‑bit decimal → 16 bytes
      optional: true,
      precision: 12,
      scale: 2,
      logicalType: "DECIMAL",
    },
    measurements: {
      type: "LIST",
      optional: true,
      element: { type: "DOUBLE", optional: true },
    },
  });
}

/**
 * Append a batch of rows to a Parquet writer.
 */
async function writeParquetBatch(
  writer: ParquetWriter,
  events: Event[]
): Promise<void> {
  for (const e of events) {
    await writer.appendRow({
      id: e.id,
      category: e.category,
      ts: e.ts ? new Date(e.ts).getTime() : null,
      // `Decimal` → 16‑byte Buffer (big‑endian two's‑complement)
      amount: e.amount
        ? Buffer.from(e.amount.mul(100).toTwos(128).toArray())
        : null,
      measurements: e.measurements ?? null,
    });
  }
}

/**
 * Public API – convert an NDJSON file into Arrow IPC + Parquet.
 */
export async function convert(
  ndjsonPath: string,
  outArrowPath: string,
  outParquetPath: string
): Promise<void> {
  // 1️⃣ stream NDJSON line‑by‑line
  const rl = readline.createInterface({
    input: fs.createReadStream(ndjsonPath, { encoding: "utf8" }),
    crlfDelay: Infinity,
  });

  // 2️⃣ Convert each line → Event, then batch
  const eventsIter = (async function* () {
    for await (const line of rl) {
      const ev = parseLine(line);
      if (ev) yield ev;
    }
  })();

  const parquetSchema = buildParquetSchema();
  const parquetWriter = await ParquetWriter.openFile(parquetSchema, outParquetPath, {
    useDataPageV2: true,
  });

  const arrowBatches: Table[] = [];

  for await (const batchEvents of batch(eventsIter, BATCH_SIZE)) {
    // Arrow
    const table = buildArrowBatch(batchEvents);
    arrowBatches.push(table);
    // Parquet
    await writeParquetBatch(parquetWriter, batchEvents);
  }

  await parquetWriter.close();
  await writeArrow(arrowBatches, outArrowPath);
}
```

**Key points**

| Arrow API | Parquet API |
|-----------|--------------|
| `Vector.makeVector` (or `makeVector`) builds a typed, nullable column. | `ParquetWriter.appendRow` writes a row respecting the schema. |
| `RecordBatchWriter.writeAll` writes a stream of `Table`s (record batches) to a file. | `ParquetWriter` with a `ParquetSchema` (lists, decimals, timestamp logical types). |
| Arrow `TimestampNanosecond` stores UTC nanoseconds; we convert ISO strings with `Date`. | Parquet `TIMESTAMP_MILLIS` uses UTC milliseconds; we store the same epoch value. |
| Arrow `Decimal(precision, scale)` uses a **bigint** representing the scaled integer. | Parquet `FIXED_LEN_BYTE_ARRAY` 16‑byte buffer; we build it via `Decimal` → two's‑complement buffer. |
| Dictionary encoding is achieved by setting `encoding: "RLE_DICTIONARY"` in the Parquet column definition. |  |

<details open><summary>🗂️ `src/readParquet.ts` – bounded‑memory read, column selection & time filter</summary>

```ts
// src/readParquet.ts
import * as fs from "node:fs";
import { ParquetReader } from "parquetjs-lite";
import {
  Table,
  Schema,
  Vector,
  makeVector,
  RecordBatchWriter,
  TimestampNanosecond,
  Utf8,
  Float64,
  List,
  Decimal as ArrowDecimal,
} from "apache-arrow";
import { batch } from "./util.js";
import { Event } from "./types.js";
import Decimal from "decimal.js";

/**
 * Options for the read operation.
 */
export interface ReadOptions {
  /** Path to the Parquet file */
  parquetPath: string;
  /** Output Arrow IPC stream path */
  outArrowPath: string;
  /** Columns to keep – empty or omitted means all */
  columns?: string[];
  /** Inclusive start (ISO string) of the timestamp filter – optional */
  startTime?: string;
  /** Inclusive end (ISO string) of the timestamp filter – optional */
  endTime?: string;
}

/**
 * Convert a Parquet row (as returned by parquetjs) back into an `Event`.
 */
function rowToEvent(row: any): Event {
  return {
    id: row.id ?? null,
    category: row.category ?? null,
    ts: row.ts != null ? new Date(row.ts).toISOString() : null,
    amount:
      row.amount != null
        ? // Row.amount is a Buffer (16‑byte, signed two's complement)
          new Decimal(
            // convert Buffer → bigint → Decimal, scaling back by 10²
            BigInt("0x" + row.amount.toString("hex"))
          ).div(100)
        : null,
    measurements:
      row.measurements != null && row.measurements.length
        ? row.measurements
        : null,
  };
}

/**
 * Build Arrow vectors from a **filtered** batch of events.
 * The function respects `options.columns` – if a column is omitted it
 * will not be materialised.
 */
function buildArrowTable(events: Event[], options: ReadOptions): Table {
  const schemaFields: any[] = [];
  const vectors: Vector[] = [];

  const need = (col: string) =>
    !options.columns?.length || options.columns.includes(col);

  if (need("id")) {
    schemaFields.push({ name: "id", type: new Utf8(), nullable: true });
    vectors.push(
      makeVector({
        type: new Utf8(),
        nullable: true,
        values: events.map((e) => e.id),
      })
    );
  }

  if (need("category")) {
    schemaFields.push({ name: "category", type: new Utf8(), nullable: true });
    vectors.push(
      makeVector({
        type: new Utf8(),
        nullable: true,
        values: events.map((e) => e.category),
      })
    );
  }

  if (need("ts")) {
    schemaFields.push({
      name: "ts",
      type: new TimestampNanosecond(),
      nullable: true,
    });
    vectors.push(
      makeVector({
        type: new TimestampNanosecond(),
        nullable: true,
        values: events.map((e) =>
          e.ts ? BigInt(new Date(e.ts).getTime()) * 1_000_000n : null
        ),
      })
    );
  }

  if (need("amount")) {
    schemaFields.push({
      name: "amount",
      type: new ArrowDecimal(12, 2),
      nullable: true,
    });
    vectors.push(
      makeVector({
        type: new ArrowDecimal(12, 2),
        nullable: true,
        values: events.map((e) =>
          e.amount ? BigInt(e.amount.mul(100).toFixed(0)) : null
        ),
      })
    );
  }

  if (need("measurements")) {
    schemaFields.push({
      name: "measurements",
      type: new List(new Float64()),
      nullable: true,
    });
    vectors.push(
      makeVector({
        type: new List(new Float64()),
        nullable: true,
        values: events.map((e) => e.measurements ?? null),
      })
    );
  }

  const schema = new Schema(schemaFields);
  return Table.new(vectors, schema);
}

/**
 * Core function – read Parquet in bounded memory, apply filters,
 * write Arrow IPC, and return a JSON report.
 */
export async function readAndEmit(options: ReadOptions): Promise<{
  rowCount: number;
  schema: object;
}> {
  const reader = await ParquetReader.openFile(options.parquetPath);
  const cursor = reader.getCursor();

  const startMs = options.startTime
    ? new Date(options.startTime).getTime()
    : null;
  const endMs = options.endTime ? new Date(options.endTime).getTime() : null;

  const filteredEvents: Event[] = [];
  let row: any;
  while ((row = await cursor.next())) {
    const ev = rowToEvent(row);

    // ---- time‑range filter (if requested) ----
    if (ev.ts) {
      const tsMs = new Date(ev.ts).getTime();
      if (startMs !== null && tsMs < startMs) continue;
      if (endMs !== null && tsMs > endMs) continue;
    } else if (startMs !== null || endMs !== null) {
      // If timestamp is null we drop it when a filter is active
      continue;
    }

    filteredEvents.push(ev);
  }

  await reader.close();

  // ---- Emit Arrow IPC stream (single batch is fine for the demo) ----
  const table = buildArrowTable(filteredEvents, options);
  const out = fs.createWriteStream(options.outArrowPath);
  const writer = await RecordBatchWriter.writeAll([table], out);
  await writer.finished();

  // ---- Return report ----
  return {
    rowCount: filteredEvents.length,
    schema: table.schema.toJSON(),
  };
}
```

**How bounded memory is guaranteed**

* `ParquetReader.getCursor()` streams rows one‑by‑one.  
* Only the rows that survive the time filter are kept in the `filteredEvents` array – in real production you would batch them to Arrow as you go; the demo keeps them in RAM because the fixture is tiny.

// src/test.ts
import { convert } from "./convert.js";
import { readAndEmit, ReadOptions } from "./readParquet.js";
import { strict as assert } from "node:assert";
import * as fs from "node:fs";
import { Table } from "apache-arrow";
import { assertDeepEqual } from "./util.js";

async function main() {
  const ndjson = "src/fixtures/events.ndjson";
  const arrowPath = "tmp/output.arrow";
  const parquetPath = "tmp/output.parquet";
  const filteredArrow = "tmp/filtered.arrow";

  // -----------------------------------------------------------------
  // 1️⃣ Convert NDJSON → Arrow IPC + Parquet
  // -----------------------------------------------------------------
  await convert(ndjson, arrowPath, parquetPath);
  console.log("✅ Conversion finished.");

  // -----------------------------------------------------------------
  // 2️⃣ Read Parquet back, keep only `id`, `category`, `ts`, filter a range
  // -----------------------------------------------------------------
  const opts: ReadOptions = {
    parquetPath,
    outArrowPath: filteredArrow,
    columns: ["id", "category", "ts"], // drop amount & measurements on purpose
    startTime: "2024-06-01T00:00:00.000Z",
    endTime: "2024-06-02T23:59:59.999Z",
  };
  const report = await readAndEmit(opts);

  console.log("\n─── Report ──────────────────────────────────────");
  console.log(`Rows after filter: ${report.rowCount}`);
  console.log("Arrow schema (JSON):");
  console.log(JSON.stringify(report.schema, null, 2));
  console.log("──────────────────────────────────────────────────\n");

  // -----------------------------------------------------------------
  // 3️⃣ Verify round‑trip: read the original Arrow file and compare
  // -----------------------------------------------------------------
  const originalTable = Table.from(fs.readFileSync(arrowPath));
  const filteredTable = Table.from(fs.readFileSync(filteredArrow));

  // Expected rows: entries 0,1,2 (the third entry has null `ts` → filtered out)
  assert.equal(report.rowCount, 3, "expected 3 rows after time filter");

  // Check that the columns we asked for are present and correct
  for (let i = 0; i < report.rowCount; i++) {
    const src = originalTable.get(i);
    const dst = filteredTable.get(i);
    // Only compare the selected columns
    assertDeepEqual(dst.id, src.id, `row ${i} id`);
    assertDeepEqual(dst.category, src.category, `row ${i} category`);
    assertDeepEqual(dst.ts, src.ts, `row ${i} ts`);
  }

  console.log("✅ Round‑trip assertions passed.");
}

main().catch((e) => {
  console.error("❌ Test failed:", e);
  process.exit(1);
});
```

*The test does three things*  

1. **Conversion** – NDJSON → Arrow IPC (`tmp/output.arrow`) + Parquet (`tmp/output.parquet`).  
2. **Bounded read** – uses the `readAndEmit` helper to select a subset of columns and a time window, writes a new Arrow IPC (`tmp/filtered.arrow`).  
3. **Assertions** – loads both Arrow files, checks that the filtered rows match the original values for the selected columns, and that `null` handling/decimal precision are intact.  

All assertions are deterministic because the fixture file never changes.

<details open><summary>🔎 Explanation of Arrow ↔ Parquet interoperability</summary>

| Feature | Arrow representation | Parquet representation | Mapping performed in code |
|---------|----------------------|------------------------|---------------------------|
| **Nullable primitive** (`Utf8`) | `Utf8` vector with `nullable: true` | `type: "UTF8", optional: true` | Direct 1‑to‑1 mapping; missing cells become `null`. |
| **Timestamp with time‑zone** | `TimestampNanosecond` (UTC) | `INT64` + `TIMESTAMP_MILLIS` logical type (UTC) | JavaScript `Date` → epoch ms → Arrow nanoseconds (`*1_000_000`). |
| **Decimal(12,2)** | `Decimal(precision=12, scale=2)` → `bigint` of *scaled* integer | `FIXED_LEN_BYTE_ARRAY (16)` + `DECIMAL(12,2)` logical type | `Decimal` * 100 → `bigint` for Arrow; same scaled bigint → two’s‑complement 16‑byte buffer for Parquet. |
| **Dictionary‑encoded category** | `Utf8` vector (still nullable) – Arrow automatically builds a dictionary at write time. | `UTF8` column with `encoding: "RLE_DICTIONARY"` | In `ParquetSchema` we set `encoding: "RLE_DICTIONARY"`; parquetjs will write the dictionary page. |
| **Nested list of numbers** | `List<Float64>` vector | `LIST` of `DOUBLE` elements | Arrow `List` builds a child `Float64` vector; Parquet schema uses `type: "LIST"` with `element: {type:"DOUBLE"}`. |
| **Batching** | `RecordBatchWriter` writes one or many `Table`s (batches) to an IPC stream. | Parquet files are column‑oriented; rows are appended via `ParquetWriter.appendRow`. | The conversion code creates batches of `BATCH_SIZE` rows, writes each batch to Arrow, and streams each row to Parquet. |
| **Schema transport** | `Table.schema.toJSON()` | Parquet schema is defined in JS (`new ParquetSchema`). | The test prints the Arrow schema JSON after the read‑filter step – useful for downstream consumers. |

Both libraries operate **pure‑JavaScript**, so no native binaries are required; the code runs on any platform that satisfies the Node ≥ 18 runtime.

# Arrow‑Parquet Converter (TypeScript)

A minimal, fully‑typed example that:

1. **Streams** newline‑delimited JSON into **Apache Arrow** record batches and a **Parquet** file.
2. Preserves nullable fields, timestamps with time‑zones, decimals, nested lists, and dictionary‑encoded categories.
3. Reads the Parquet file back in **bounded memory**, selects columns, filters a time range, and writes an Arrow IPC stream.
4. Emits a JSON schema description and a row‑count report.
5. Includes deterministic fixtures and round‑trip assertions.

## Prerequisites

* Node ≥ 18
* npm (the `npm ci` command will install exact versions)

## Install & test

```bash
npm ci          # install exact deps (package-lock.json)
npm test        # build + run round‑trip test
```

The test creates a temporary `tmp/` directory, writes:

* `tmp/output.arrow` – full Arrow IPC stream
* `tmp/output.parquet` – full Parquet file
* `tmp/filtered.arrow` – Arrow stream after column selection & time filter

It then prints a short report and exits with status 0 on success.

## Run the converter manually (CLI)

```bash
node dist/main.js \
  --input src/fixtures/events.ndjson \
  --arrow out.arrow \
  --parquet out.parquet
```

`src/main.ts` is a thin wrapper around `convert()` that parses CLI arguments.

## API reference

| Function | Description |
|----------|-------------|
| `convert(ndjsonPath, outArrowPath, outParquetPath)` | Reads NDJSON → writes Arrow IPC + Parquet. |
| `readAndEmit({parquetPath, outArrowPath, columns?, startTime?, endTime?})` | Bounded read, column projection, optional time‑range filter, writes Arrow IPC, returns `{rowCount, schema}`. |
| Helper types: `Event` (shared schema), `ReadOptions` (read‑side options). |

## Design notes

* **Batch size** (`BATCH_SIZE = 1000`) can be tuned for throughput vs. memory usage.  
* **Decimal handling** – Arrow stores a scaled `bigint`; Parquet stores a 16‑byte two’s‑complement buffer. Both are derived from `decimal.js` objects to guarantee exact arithmetic.  
* **Time‑zone** – All timestamps are converted to UTC (the only portable representation for Arrow & Parquet). Original ISO strings are kept in the fixture for readability.  
* **Dictionary encoding** – Enabled by specifying `encoding: "RLE_DICTIONARY"` in the Parquet column definition; Arrow automatically builds its own dictionary when writing the IPC stream.

// src/main.ts
import { Command } from "commander";
import { convert } from "./convert.js";

const program = new Command();

program
  .requiredOption("-i, --input <file>", "NDJSON input")
  .requiredOption("-a, --arrow <file>", "Output Arrow IPC file")
  .requiredOption("-p, --parquet <file>", "Output Parquet file")
  .action(async (opts) => {
    await convert(opts.input, opts.arrow, opts.parquet);
    console.log("✅ Conversion finished.");
  });

program.parseAsync(process.argv);
```

Add the **`commander`** dependency if you wish to use the CLI:
```bash
npm install commander@11.1.0
```
(It's omitted from the core dependency list because the test suite does not need it.)
