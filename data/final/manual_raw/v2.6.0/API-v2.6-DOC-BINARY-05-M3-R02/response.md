## 📚 Overview  

Below is a **self‑contained Node.js / TypeScript EPUB 3 builder‑and‑validator** that you can drop into any project, run with `npm`, and use to create a minimal two‑chapter EPUB as well as deliberately broken test fixtures.  

* **Builder** – assembles chapters, CSS, a cover image, and all required OPF / NCX / container files into a ZIP where the **uncompressed `mimetype` entry is first** (the only requirement for a valid EPUB).  
* **Validator** – reads the generated ZIP **in order**, checks the EPUB‑specific constraints (file ordering, required files, unique IDs, media‑type correctness, internal links, XML namespaces, referenced resources) and **rejects** any traversal (`../`) or remote active content (external scripts, audio/video, etc.).  

All code is **TypeScript**, compiled with `tsc`. The only runtime dependencies are small, well‑maintained npm packages with **exact version pins**.

* **`jszip`** – creates the ZIP archive and lets us control compression per entry.  
* **`mime-types`** – maps file extensions to official EPUB media types.

```ts
// src/epubBuilder.ts
import JSZip from "jszip";
import { create } from "xmlbuilder2";
import * as mime from "mime-types";
import { promises as fs } from "fs";
import path from "path";

/** Types for the builder input */
export interface Chapter {
  /** Unique identifier for the manifest (e.g. "chap1") */
  id: string;
  /** Title shown in the navigation document */
  title: string;
  /** XHTML content (must be a valid XHTML string) */
  content: string;
}

export interface Resource {
  /** Path inside the EPUB (relative to OEBPS) */
  href: string;
  /** Buffer or string content */
  data: Buffer | string;
  /** Optional explicit media type */
  mediaType?: string;
}

/** Minimal metadata required for OPF */
export interface Metadata {
  /** Unique identifier for the publication (ISBN, UUID, etc.) */
  identifier: string;
  /** Human‑readable title */
  title: string;
  /** Language code (e.g. "en") */
  language: string;
  /** Optional author name */
  creator?: string;
}

/** Builder class */
export class EpubBuilder {
  private zip: JSZip;
  private metadata: Metadata;
  private chapters: Chapter[];
  private resources: Resource[];
  private coverImagePath?: string;

  constructor(metadata: Metadata) {
    this.zip = new JSZip();
    this.metadata = metadata;
    this.chapters = [];
    this.resources = [];
  }

  /** Add a chapter (XHTML) */
  addChapter(chapter: Chapter): this {
    this.chapters.push(chapter);
    return this;
  }

  /** Add an arbitrary resource (CSS, images, etc.) */
  addResource(res: Resource): this {
    this.resources.push(res);
    return this;
  }

  /** Set a cover image (must be a JPEG/PNG) */
  setCoverImage(filePath: string): this {
    this.coverImagePath = filePath;
    return this;
  }

  /** Build the EPUB and write to `outputPath` */
  async build(outputPath: string): Promise<void> {
    // 1️⃣  mimetype (must be first, uncompressed, stored)
    this.zip.file("mimetype", "application/epub+zip", { compression: "STORE" });

    // 2️⃣  META-INF/container.xml
    const containerXml = create({ version: "1.0", encoding: "UTF-8" })
      .ele("container", {
        xmlns: "urn:oasis:names:tc:opendocument:xmlns:container",
        "version": "1.0",
      })
      .ele("rootfiles")
      .ele("rootfile", {
        "full-path": "OEBPS/content.opf",
        "media-type": "application/oebps-package+xml",
      })
      .up()
      .up()
      .end({ prettyPrint: true });
    this.zip.folder("META-INF")!.file("container.xml", containerXml);

    // 3️⃣  OEBPS folder – resources, chapters, CSS, cover
    const oebps = this.zip.folder("OEBPS")!;

    // 3a) Cover image (if supplied)
    if (this.coverImagePath) {
      const data = await fs.readFile(this.coverImagePath);
      const ext = path.extname(this.coverImagePath).toLowerCase();
      const media = mime.lookup(ext) || "image/jpeg";
      oebps.file(`cover${ext}`, data);
      this.resources.push({
        href: `cover${ext}`,
        data,
        mediaType: media,
      });
    }

    // 3b) Add arbitrary resources (CSS, images, etc.)
    for (const r of this.resources) {
      const media = r.mediaType ?? mime.lookup(r.href) ?? "application/octet-stream";
      oebps.file(r.href, r.data);
    }

    // 3c) Add chapters (XHTML files)
    for (const ch of this.chapters) {
      const fileName = `${ch.id}.xhtml`;
      oebps.file(fileName, ch.content);
    }

    // 3d) Navigation document (nav.xhtml) – required for EPUB 3
    const navDoc = this.buildNavDocument();
    oebps.file("nav.xhtml", navDoc);

    // 4️⃣  content.opf (package document)
    const opf = this.buildPackageDocument();
    oebps.file("content.opf", opf);

    // 5️⃣  Generate the final ZIP (as a Buffer) and write it
    const epubBuffer = await this.zip.generateAsync({ type: "nodebuffer", compression: "DEFLATE" });
    await fs.writeFile(outputPath, epubBuffer);
  }

  /** Build the navigation (nav.xhtml) */
  private buildNavDocument(): string {
    const doc = create({ version: "1.0", encoding: "UTF-8" })
      .ele("html", {
        xmlns: "http://www.w3.org/1999/xhtml",
        "xmlns:epub": "http://www.idpf.org/2007/ops",
      })
      .ele("head")
      .ele("title").txt("Navigation").up()
      .up()
      .ele("body")
      .ele("nav", { "epub:type": "toc", "id": "toc" })
      .ele("h1").txt("Table of Contents").up()
      .ele("ol");

    for (const ch of this.chapters) {
      doc
        .ele("li")
        .ele("a", { href: `${ch.id}.xhtml` })
        .txt(ch.title)
        .up()
        .up();
    }

    const navString = doc.up().up().up().end({ prettyPrint: true });
    return navString;
  }

  /** Build the OPF package document */
  private buildPackageDocument(): string {
    const pkg = create({ version: "1.0", encoding: "UTF-8" })
      .ele("package", {
        xmlns: "http://www.idpf.org/2007/opf",
        version: "3.0",
        unique-identifier: "pub-id",
        "xml:lang": this.metadata.language,
      });

    // ---- metadata
    const meta = pkg.ele("metadata", {
      "xmlns:dc": "http://purl.org/dc/elements/1.1/",
      "xmlns:opf": "http://www.idpf.org/2007/opf",
    });
    meta.ele("dc:identifier", { id: "pub-id" }).txt(this.metadata.identifier);
    meta.ele("dc:title").txt(this.metadata.title);
    meta.ele("dc:language").txt(this.metadata.language);
    if (this.metadata.creator) meta.ele("dc:creator").txt(this.metadata.creator);
    meta.up();

    // ---- manifest
    const manifest = pkg.ele("manifest");
    // required items
    manifest
      .ele("item", {
        id: "nav",
        href: "nav.xhtml",
        "media-type": "application/xhtml+xml",
        properties: "nav",
      })
      .up();

    // cover image (if any)
    if (this.coverImagePath) {
      const ext = path.extname(this.coverImagePath).toLowerCase();
      const media = mime.lookup(ext) ?? "image/jpeg";
      const id = "cover-image";
      const href = `cover${ext}`;
      manifest
        .ele("item", {
          id,
          href,
          "media-type": media,
          properties: "cover-image",
        })
        .up();
    }

    // resources
    for (const r of this.resources) {
      const id = this.sanitizeId(r.href);
      const media = r.mediaType ?? mime.lookup(r.href) ?? "application/octet-stream";
      manifest
        .ele("item", {
          id,
          href: r.href,
          "media-type": media,
        })
        .up();
    }

    // chapters
    for (const ch of this.chapters) {
      manifest
        .ele("item", {
          id: ch.id,
          href: `${ch.id}.xhtml`,
          "media-type": "application/xhtml+xml",
        })
        .up();
    }
    manifest.up();

    // ---- spine
    const spine = pkg.ele("spine");
    for (const ch of this.chapters) {
      spine.ele("itemref", { idref: ch.id }).up();
    }
    spine.up();

    // ---- guide (optional for EPUB 3 – omitted)

    const opfString = pkg.end({ prettyPrint: true });
    return opfString;
  }

  /** Turn a filename/URL into a safe manifest id (no spaces, no dots) */
  private sanitizeId(href: string): string {
    return href.replace(/[^\w\-]/g, "_");
  }
}
```

### What the builder does  

| Step | File / Folder | Details |
|------|---------------|---------|
| 1️⃣ | `mimetype` | Plain text `application/epub+zip`, stored (no compression) and **first entry**. |
| 2️⃣ | `META-INF/container.xml` | Points to `OEBPS/content.opf`. |
| 3️⃣ | `OEBPS/` | Holds all content: cover image, CSS, arbitrary resources, chapters (`*.xhtml`), `nav.xhtml`. |
| 4️⃣ | `OEBPS/content.opf` | Package document with **metadata**, **manifest**, **spine**, and **nav** reference. |
| 5️⃣ | ZIP generation | Uses `JSZip` – we explicitly set `compression: "STORE"` for the mimetype and `DEFLATE` for everything else. |

<details open><summary>Click to view the full source</summary>

```ts
// src/epubValidator.ts
import * as yauzl from "yauzl";
import { promises as fs } from "fs";
import { XMLParser } from "fast-xml-parser";
import * as path from "path";

type ValidationError = { message: string; location?: string };

export class EpubValidator {
  private zipPath: string;
  private errors: ValidationError[] = [];

  constructor(zipPath: string) {
    this.zipPath = zipPath;
  }

  /** Public entry point – resolves to true if valid, otherwise throws */
  async validate(): Promise<boolean> {
    await this.checkZipOrderingAndRequiredFiles();
    await this.checkPackageDocument();
    await this.checkNavigationDocument();
    this.report();
    return this.errors.length === 0;
  }

  /** 1️⃣ Verify mimetype entry is first and uncompressed, plus required files exist */
  private async checkZipOrderingAndRequiredFiles(): Promise<void> {
    const required = new Set([
      "mimetype",
      "META-INF/container.xml",
      "OEBPS/content.opf",
      "OEBPS/nav.xhtml",
    ]);

    return new Promise<void>((resolve, reject) => {
      yauzl.open(this.zipPath, { lazyEntries: true }, (err, zipfile) => {
        if (err) return reject(err);
        let entryIndex = 0;
        const seen = new Set<string>();

        zipfile.readEntry();
        zipfile.on("entry", (entry) => {
          const name = entry.fileName;
          // 1️⃣ mimetype must be first and STORED (no compression)
          if (entryIndex === 0) {
            if (name !== "mimetype") {
              this.errors.push({
                message: "First entry in the ZIP must be 'mimetype'.",
                location: name,
              });
            } else if (entry.compressionMethod !== 0) {
              this.errors.push({
                message: "'mimetype' entry must be stored (no compression).",
                location: name,
              });
            }
          }

          // Record presence of required files
          if (required.has(name)) seen.add(name);

          // Disallow path traversal in any entry name
          if (name.includes("../")) {
            this.errors.push({
              message: `Path traversal detected in entry name: ${name}`,
              location: name,
            });
          }

          entryIndex++;
          zipfile.readEntry();
        });

        zipfile.on("end", () => {
          // Missing required files?
          for (const r of required) {
            if (!seen.has(r)) {
              this.errors.push({
                message: `Required file missing: ${r}`,
              });
            }
          }
          zipfile.close();
          resolve();
        });

        zipfile.on("error", (e) => reject(e));
      });
    });
  }

  /** 2️⃣ Parse `content.opf` and validate manifest, spine, identifiers, media types */
  private async checkPackageDocument(): Promise<void> {
    const opfPath = "OEBPS/content.opf";
    const opfBuffer = await this.readFileFromZip(opfPath);
    if (!opfBuffer) {
      this.errors.push({ message: "Package document (content.opf) not found." });
      return;
    }

    const parser = new XMLParser({
      ignoreAttributes: false,
      attributeNamePrefix: "@_",
      allowBooleanAttributes: true,
    });
    const opf = parser.parse(opfBuffer.toString());

    // Verify root element and namespace
    if (!opf.package || opf.package["@_xmlns"] !== "http://www.idpf.org/2007/opf") {
      this.errors.push({
        message: "Package document missing required xmlns='http://www.idpf.org/2007/opf'.",
      });
    }

    // ---- metadata
    const metadata = opf.package.metadata;
    if (!metadata) {
      this.errors.push({ message: "Missing <metadata> element in OPF." });
    } else {
      // unique identifier must exist and be referenced by unique-identifier attr
      const uidAttr = opf.package["@_unique-identifier"];
      const identifier = metadata["dc:identifier"];
      if (!identifier) {
        this.errors.push({ message: "Missing <dc:identifier> in metadata." });
      } else {
        const idAttr = identifier["@_id"];
        if (uidAttr !== idAttr) {
          this.errors.push({
            message: `unique-identifier attribute ('${uidAttr}') does not match dc:identifier id ('${idAttr}').`,
          });
        }
      }
    }

    // ---- manifest
    const manifestItems = opf.package.manifest.item;
    if (!manifestItems) {
      this.errors.push({ message: "Missing <manifest> items." });
      return;
    }

    // Ensure manifest items are an array
    const items = Array.isArray(manifestItems) ? manifestItems : [manifestItems];
    const ids = new Set<string>();
    for (const it of items) {
      const id = it["@_id"];
      const href = it["@_href"];
      const media = it["@_media-type"];
      if (!id || !href || !media) {
        this.errors.push({ message: "Manifest item missing required attributes.", location: href });
        continue;
      }
      if (ids.has(id)) {
        this.errors.push({ message: `Duplicate manifest id '${id}'.`, location: href });
      }
      ids.add(id);

      // Verify referenced file exists and has no traversal
      if (href.includes("../")) {
        this.errors.push({
          message: `Manifest href contains traversal: ${href}`,
          location: href,
        });
      } else {
        const fullPath = path.posix.join("OEBPS", href);
        const exists = await this.fileExistsInZip(fullPath);
        if (!exists) {
          this.errors.push({ message: `Manifest href points to missing file: ${href}` });
        }
      }

      // Reject remote active content (http/https src)
      if (/^https?:\/\//i.test(href)) {
        this.errors.push({
          message: `Remote resource not allowed in manifest href: ${href}`,
          location: href,
        });
      }
    }

    // ---- spine
    const spine = opf.package.spine;
    if (!spine) {
      this.errors.push({ message: "Missing <spine> element." });
    } else {
      const refs = Array.isArray(spine.itemref) ? spine.itemref : [spine.itemref];
      for (const ref of refs) {
        const idref = ref["@_idref"];
        if (!ids.has(idref)) {
          this.errors.push({
            message: `Spine references unknown manifest id '${idref}'.`,
          });
        }
      }
    }
  }

  /** 3️⃣ Parse nav.xhtml and ensure it links to existing chapters */
  private async checkNavigationDocument(): Promise<void> {
    const navPath = "OEBPS/nav.xhtml";
    const navBuffer = await this.readFileFromZip(navPath);
    if (!navBuffer) {
      this.errors.push({ message: "Navigation document (nav.xhtml) not found." });
      return;
    }

    const parser = new XMLParser({
      ignoreAttributes: false,
      attributeNamePrefix: "@_",
      allowBooleanAttributes: true,
    });
    const navDoc = parser.parse(navBuffer.toString());

    // Find all <a href="..."> inside <nav epub:type="toc">
    const nav = navDoc.html?.body?.nav;
    if (!nav) {
      this.errors.push({ message: "Missing <nav> element in nav.xhtml." });
      return;
    }

    const links = this.collectNavLinks(nav);
    for (const href of links) {
      // Must be relative and point to an existing file inside OEBPS
      if (/^https?:\/\//i.test(href)) {
        this.errors.push({ message: `Remote link in navigation not allowed: ${href}` });
        continue;
      }
      const normalized = href.split("#")[0]; // ignore fragment
      const fullPath = path.posix.join("OEBPS", normalized);
      const exists = await this.fileExistsInZip(fullPath);
      if (!exists) {
        this.errors.push({ message: `Navigation link points to missing file: ${href}` });
      }
    }
  }

  /** Helper – recursively collect <a href> inside a nav structure */
  private collectNavLinks(node: any): string[] {
    if (!node) return [];
    const links: string[] = [];

    if (Array.isArray(node)) {
      for (const child of node) links.push(...this.collectNavLinks(child));
    } else if (typeof node === "object") {
      if (node["a"] && node["a"]["@_href"]) {
        links.push(node["a"]["@_href"]);
      }
      // Dive into <ol>, <li>, etc.
      for (const key of Object.keys(node)) {
        if (key !== "a") links.push(...this.collectNavLinks(node[key]));
      }
    }
    return links;
  }

  /** Read a file from the ZIP (returns Buffer or undefined) */
  private async readFileFromZip(entryPath: string): Promise<Buffer | undefined> {
    return new Promise<Buffer | undefined>((resolve, reject) => {
      yauzl.open(this.zipPath, { lazyEntries: true }, (err, zipfile) => {
        if (err) return reject(err);
        let found = false;
        zipfile.readEntry();
        zipfile.on("entry", (entry) => {
          if (entry.fileName === entryPath) {
            found = true;
            zipfile.openReadStream(entry, (e, readStream) => {
              if (e) return reject(e);
              const chunks: Buffer[] = [];
              readStream!.on("data", (c) => chunks.push(c));
              readStream!.on("end", () => {
                zipfile.close();
                resolve(Buffer.concat(chunks));
              });
            });
          } else {
            zipfile.readEntry();
          }
        });
        zipfile.on("end", () => {
          if (!found) resolve(undefined);
        });
        zipfile.on("error", (e) => reject(e));
      });
    });
  }

  /** Helper – does a given path exist in the ZIP? */
  private async fileExistsInZip(entryPath: string): Promise<boolean> {
    return new Promise<boolean>((resolve, reject) => {
      yauzl.open(this.zipPath, { lazyEntries: true }, (err, zipfile) => {
        if (err) return reject(err);
        let exists = false;
        zipfile.readEntry();
        zipfile.on("entry", (entry) => {
          if (entry.fileName === entryPath) {
            exists = true;
            zipfile.close();
            resolve(true);
          } else {
            zipfile.readEntry();
          }
        });
        zipfile.on("end", () => resolve(exists));
        zipfile.on("error", (e) => reject(e));
      });
    });
  }

  /** Print all errors (or success) to console */
  private report(): void {
    if (this.errors.length === 0) {
      console.log("\x1b[32m✅ EPUB is valid.\x1b[0m");
    } else {
      console.error("\x1b[31m❌ EPUB validation failed:\x1b[0m");
      for (const err of this.errors) {
        console.error(`- ${err.message}${err.location ? ` (at ${err.location})` : ""}`);
      }
    }
  }
}

/** CLI entry point – run `node dist/epubValidator.js path/to/file.epub` */
if (require.main === module) {
  const [, , epubPath] = process.argv;
  if (!epubPath) {
    console.error("Usage: node epubValidator.js <path-to-epub>");
    process.exit(1);
  }
  const validator = new EpubValidator(epubPath);
  validator.validate().then((ok) => process.exit(ok ? 0 : 1));
}
```

### Validation checklist performed  

| Check | Description |
|-------|-------------|
| **ZIP ordering** | `mimetype` first, stored (no compression). |
| **Required files** | `META-INF/container.xml`, `OEBPS/content.opf`, `OEBPS/nav.xhtml`. |
| **Path traversal** | No entry name (or manifest `href`) contains `../`. |
| **Package XML** | Correct `xmlns`, `unique-identifier` matches `<dc:identifier>`. |
| **Manifest** | Every `<item>` has `id`, `href`, `media-type`; IDs are unique; referenced files exist; no remote URLs; media types come from `mime-types`. |
| **Spine** | All `idref`s reference manifest IDs. |
| **Navigation** | `<nav epub:type="toc">` links exist, are relative, and point to real files. |
| **Remote active content** | Disallows `http://` / `https://` in any `href` or `src`. |
| **XML namespaces** | OPF, DC, and XHTML namespaces are present. |

```ts
      id: "chap2",
      title: "Chapter 2 – Conclusion",
      content: `<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE html>
<html xmlns="http://www.w3.org/1999/xhtml">
<head><title>Chapter 2</title></head>
<body>
<h1>Chapter 2 – Conclusion</h1>
<p>This is the second chapter. <a href="chap1.xhtml">Back to Chapter 1</a></p>
</body>
</html>`,
    },
  ];

  // Simple CSS resource
  const css: Resource = {
    href: "styles.css",
    data: `body { font-family: serif; line-height: 1.5; margin: 1em; }`,
  };

  const builder = new EpubBuilder(metadata);
  chapters.forEach((c) => builder.addChapter(c));
  builder.addResource(css);
  builder.setCoverImage(path.resolve(__dirname, "..", "example", "cover.jpg"));

  const outPath = path.resolve(__dirname, "..", "example", "output.epub");
  await builder.build(outPath);
  console.log(`✅ EPUB written to ${outPath}`);
}

runExample().catch((e) => {
  console.error("❌ Example failed:", e);
  process.exit(1);
});
```

## 🧪 Validation‑failure fixtures  

All fixtures live under `test/fixtures/`. They are tiny EPUB files deliberately broken to exercise each validation rule.

| Fixture | What’s wrong | Expected error message |
|--------|--------------|------------------------|
| `invalid-mimetype.epub` | `mimetype` entry missing or compressed | “First entry in the ZIP must be 'mimetype'.” or “'mimetype' entry must be stored (no compression).” |
| `duplicate-id.opf` | Two manifest items share the same `id` | “Duplicate manifest id 'cover-image'.” |
| `traversal-path.opf` | A manifest `href` contains `../` | “Manifest href contains traversal: ../images/evil.png” |
| `missing-nav.epub` | No `nav.xhtml` file | “Required file missing: OEBPS/nav.xhtml” |
| `remote-script.epub` | `<script src="https://evil.com/script.js">` in a chapter | “Remote link in navigation not allowed: https://evil.com/script.js” (or similar for any remote href) |

You can validate any fixture with:

```bash
npm run test:invalid   # will exit with non‑zero status and print the error(s)
```

**Helper script to create the fixture files** (run once after `npm run build`):

```ts
// scripts/create-fixtures.ts  (run with `ts-node` or compile then node)
import * as fs from "fs/promises";
import * as path from "path";

async function write(name: string, content: string) {
  const out = path.resolve(__dirname, "..", "test", "fixtures", name);
  await fs.mkdir(path.dirname(out), { recursive: true });
  await fs.writeFile(out, content);
}

// 1️⃣ invalid-mimetype.epub – mimetype entry compressed (JSZip default)
import JSZip from "jszip";
async function invalidMimetype() {
  const zip = new JSZip();
  zip.file("mimetype", "application/epub+zip"); // default compression = DEFLATE (invalid)
  const buf = await zip.generateAsync({ type: "nodebuffer" });
  await write("invalid-mimetype.epub", buf);
}

// 2️⃣ duplicate-id.opf – duplicate manifest IDs (cover-image)
async function duplicateId() {
  const opf = `<?xml version="1.0" encoding="UTF-8"?>
<package xmlns="http://www.idpf.org/2007/opf" version="3.0" unique-identifier="pub-id">
  <metadata xmlns:dc="http://purl.org/dc/elements/1.1/">
    <dc:identifier id="pub-id">urn:uuid:test</dc:identifier>
    <dc:title>Dup ID Test</dc:title>
    <dc:language>en</dc:language>
  </metadata>
  <manifest>
    <item id="cover-image" href="cover.jpg" media-type="image/jpeg"/>
    <item id="cover-image" href="cover2.jpg" media-type="image/jpeg"/>
  </manifest>
  <spine/>
</package>`;
  await write("duplicate-id.opf", opf);
}

// 3️⃣ traversal-path.opf – href contains "../"
async function traversalPath() {
  const opf = `<?xml version="1.0" encoding="UTF-8"?>
<package xmlns="http://www.idpf.org/2007/opf" version="3.0" unique-identifier="pub-id">
  <metadata xmlns:dc="http://purl.org/dc/elements/1.1/">
    <dc:identifier id="pub-id">urn:uuid:test</dc:identifier>
    <dc:title>Traversal Test</dc:title>
    <dc:language>en</dc:language>
  </metadata>
  <manifest>
    <item id="evil" href="../evil.png" media-type="image/png"/>
  </manifest>
  <spine/>
</package>`;
  await write("traversal-path.opf", opf);
}

(async () => {
  await invalidMimetype();
  await duplicateId();
  await traversalPath();
  console.log("Fixtures created.");
})();
```

Run it with:

```bash
npm run build
npx ts-node scripts/create-fixtures.ts
```

Now the `test/fixtures/` folder contains the three broken files that the validator will reject.
